## Approach

**`describe.each` parameterization pattern** — already established in `src/adapters/source/conformance.test.ts:78-95` (per RESEARCH §7). The new section plugs in alongside the existing adapter cases. Skeleton:

```ts
const adapters: AdapterCase[] = [
  { name: "obsidian-fs", buildSource: (...) => new ObsidianFsSource(...), buildIndex: indexAtlasRobotics },
  { name: "stub-assembly", buildSource: () => new StubSource(makeAssemblyStubDocs()), buildIndex: indexStubAssembly },
];

describe.each(adapters)("Assembly tools — $name", (c) => {
  // Setup: build vault DB, register sources, run indexer to populate sections.
  let vault: Vault;
  beforeAll(async () => {
    vault = await c.buildIndex();  // creates in-memory DB, indexes the fixture
  });

  it("get_outline returns nested tree with anchors", async () => {
    const out = await getOutline(deps, { doc_id: c.longDocId });
    expect(out.root).toHaveLength(...);   // root sections per fixture
    expect(out.root[0].anchor).toMatch(/^[a-f0-9]{64}$/);
    // ...further structural assertions...
  });

  it("assemble_dossier matches type=Person + alias key", async () => {
    const d = await assembleDossier(deps, { type: "Person", key: "Alice C." });
    expect(d.anchor.title).toBe("Alice");
    expect(d.linked_documents.length).toBeGreaterThan(0);
    for (const l of d.linked_documents) {
      expect(l.relation).toBe("wikilink");  // v2.0.0 limitation
      expect(typeof l.properties).toBe("object");  // CitationPacket.properties is REQUIRED, never undefined
    }
  });

  it("search_sections promotes chunks to sections by max RRF", async () => { ... });
  it("get_document_bundle includes recent_edits up to 10", async () => { ... });
  it("citation packet shape is byte-identical to recall", async () => {
    const recallHit = await recall(deps, { query: "..." });
    const dossierHit = (await assembleDossier(deps, { type: "Person", key: "Alice" })).linked_documents[0];
    const recallPacketKeys = new Set(Object.keys(recallHit.results[0]));
    const dossierPacketKeys = new Set(Object.keys(dossierHit));
    // All 8 fields REQUIRED per src/memory/citation-packet.ts:45-62 (D-01 from Phase 2).
    // properties is REQUIRED — always at least {} when no frontmatter — NOT optional.
    const REQUIRED = ["doc_id", "source_handle", "title", "heading_path", "mtime", "hash", "display_url", "properties"];
    for (const k of REQUIRED) {
      expect(recallPacketKeys.has(k)).toBe(true);
      expect(dossierPacketKeys.has(k)).toBe(true);
    }
    // Type assertion: properties is an object, never undefined.
    expect(typeof recallHit.results[0].properties).toBe("object");
    expect(typeof dossierHit.properties).toBe("object");
  });
});
```
