    ```typescript
    // vault-memory:claude-ok — process state (~/.vault-memory/locks/) not vault content.
    // See ADR-005 §"Lockfile carve-out" for the rationale.
    //
    // This file is the ONLY exemption to scripts/lint-adapters.sh in Phase 5.
    // Adapter-seam discipline (no fs/path.join outside src/adapters/*/) does NOT
    // apply: lockfiles are process state managed by ~/.vault-memory/, not user
    // vault content. Per D-08 + ADR-005.

    import { open, readFile, unlink, mkdir } from "node:fs/promises";
    import { homedir } from "node:os";
    import { join } from "node:path";

    export interface LockAcquired {
      acquired: true;
      pid: number;
      path: string;
      stolenFromPid?: number;
    }

    export interface LockContended {
      acquired: false;
      ownerPid: number;
      path: string;
    }

    export type LockResult = LockAcquired | LockContended;

    function lockDir(): string {
      return join(homedir(), ".vault-memory", "locks");
    }

    function lockPath(vaultName: string): string {
      return join(lockDir(), `${vaultName}.lock`);
    }

    /** POSIX kill(pid, 0): returns true if pid is alive, false on ESRCH. */
    export function isProcessAlive(pid: number): boolean {
      try {
        process.kill(pid, 0);
        return true;
      } catch (err) {
        // ESRCH means no such process. EPERM means alive but inaccessible.
        if ((err as NodeJS.ErrnoException).code === "ESRCH") return false;
        // Defensive: any other error treated as alive (we won't steal).
        return true;
      }
    }

    async function readOwnerPid(path: string): Promise<number | null> {
      try {
        const buf = await readFile(path, "utf8");
        const pid = parseInt(buf.trim(), 10);
        return Number.isFinite(pid) && pid > 0 ? pid : null;
      } catch {
        return null;
      }
    }

    /**
     * Try to acquire the lock for a vault.
     * Atomic create via fs.open(path, 'wx') ('wx' = O_WRONLY | O_CREAT | O_EXCL).
     * On EEXIST: read current owner PID; if dead (ESRCH), steal the lock; else return contended.
     */
    export async function tryAcquireLock(vaultName: string): Promise<LockResult> {
      const dir = lockDir();
      await mkdir(dir, { recursive: true });
      const path = lockPath(vaultName);

      const attempt = async (stolenFromPid?: number): Promise<LockResult> => {
        try {
          const handle = await open(path, "wx");
          await handle.writeFile(`${process.pid}\n`);
          await handle.close();
          const result: LockAcquired = { acquired: true, pid: process.pid, path };
          if (stolenFromPid != null) result.stolenFromPid = stolenFromPid;
          return result;
        } catch (err) {
          if ((err as NodeJS.ErrnoException).code !== "EEXIST") throw err;
          const ownerPid = await readOwnerPid(path);
          if (ownerPid == null || !isProcessAlive(ownerPid)) {
            // Stale lock: remove and retry (defensive: don't loop forever).
            await unlink(path).catch(() => undefined);
            return attempt(ownerPid ?? -1);
          }
          return { acquired: false, ownerPid, path };
        }
      };

      return attempt();
    }

    /** Release the lock if we hold it. Safe to call even if we don't own it. */
    export async function releaseLock(vaultName: string): Promise<void> {
      await unlink(lockPath(vaultName)).catch(() => undefined);
    }
    ```

    ```typescript
    import type { ChangeFeed, ChangeEvent, Disposable } from "../adapters/change-feed/types.js";
    import type { DeliveryAdapter } from "../adapters/delivery/types.js";
    import type { SourceConnector } from "../adapters/source/types.js";
    import type { MemorySinkRegistry } from "../memory/registry.js";
    import type { Vault } from "../vault/index.js";
    import type { DocId } from "../types.js";
    import { recomputeCurrentHash } from "./source-hashes.js";
    import { tryAcquireLock, releaseLock } from "./lock.js";

    interface DaemonDeps {
      memorySinkRegistry: MemorySinkRegistry;
      deliveryAdapterFor: (vaultName: string) => DeliveryAdapter;
      sourceConnectorFor: (vaultName: string) => SourceConnector;
    }

    interface PendingDelete {
      id: DocId;
      chunkHashes: Set<string>;  // sha256:... of every chunk on the deleted doc, captured at delete time
      timestamp: number;
    }

    export class BriefStalenessDaemon {
      private disposable: Disposable | null = null;
      private vault!: Vault;
      private deps!: DaemonDeps;
      private acquired = false;
      private pendingDeletes = new Map<DocId, PendingDelete>();  // BRF-08 rename grace-window

      async start(vault: Vault, feed: ChangeFeed, deps: DaemonDeps): Promise<void> {
        this.vault = vault;
        this.deps = deps;

        const lock = await tryAcquireLock(vault.config.name);
        if (!lock.acquired) {
          // D-08: structured WARN + return early. Multi-client friendly.
          vault.db.audit.recordWrite({
            kind: "daemon_already_owned",
            doc_id: null,
            vault_name: vault.config.name,
            details: { ownerPid: lock.ownerPid, path: lock.path },
          });
          // stderr WARN for human-visible diagnostic (CONTEXT D-08).
          // eslint-disable-next-line no-console
          console.warn(`[vault-memory] brief staleness daemon already owned by PID ${lock.ownerPid} — briefs will be marked stale only by the other process.`);
          return;
        }
        this.acquired = true;

        // Diagnostic: log starting cursor for "is my daemon current?" introspection.
        const startCursor = vault.db.daemonState.getCursor(vault.config.name);

        // Step 1: Startup full scan (D-09 correctness floor).
        await this.runStartupScan();

        // Step 2: Subscribe to ChangeFeed (D-07).
        this.disposable = feed.subscribe(async (event: ChangeEvent) => {
          try {
            await this.handleEvent(event);
            vault.db.daemonState.setCursor(vault.config.name, Date.now());
          } catch (err) {
            const message = err instanceof Error ? err.message : String(err);
            vault.db.audit.recordWrite({
              kind: "brief_staleness_error",
              doc_id: null,
              vault_name: vault.config.name,
              details: { event_kind: event.kind, event_id: "id" in event ? event.id : null, message },
            });
          }
        });

        // Set initial cursor.
        vault.db.daemonState.setCursor(vault.config.name, Date.now());
      }

      async shutdown(): Promise<void> {
        // Dispose subscription FIRST so no more events arrive.
        if (this.disposable) {
          this.disposable[Symbol.dispose]();
          this.disposable = null;
        }
        // Release lock LAST. A crashed shutdown that fails here leaves the
        // lock for stale-detection (kill(pid, 0)) to recover on next boot.
        if (this.acquired) {
          await releaseLock(this.vault.config.name);
          this.acquired = false;
        }
      }

      private async runStartupScan(): Promise<void> {
        const briefIds = this.vault.db.briefSources.listBriefDocIds();
        for (const briefId of briefIds) {
          try {
            await this.evaluateBrief(briefId);
          } catch (err) {
            const message = err instanceof Error ? err.message : String(err);
            this.vault.db.audit.recordWrite({
              kind: "brief_staleness_error",
              doc_id: briefId,
              vault_name: this.vault.config.name,
              details: { phase: "startup_scan", message },
            });
          }
        }
      }

      private async handleEvent(event: ChangeEvent): Promise<void> {
        switch (event.kind) {
          case "create":
            await this.handleCreate(event.id);
            break;
          case "update":
            await this.evaluateChangedDocId(event.id);
            break;
          case "delete":
            await this.handleDelete(event.id);
            break;
          case "rename":
            await this.handleRenameDirect(event.old_id, event.new_id);
            break;
        }
        // Tick grace-window expiries on every event.
        this.expireGraceWindow();
      }

      private async evaluateChangedDocId(docId: DocId): Promise<void> {
    ```
