    function walkMarkdown(root: string): { path: string; fm: Record<string, unknown>; body: string }[] {
      const out: { path: string; fm: Record<string, unknown>; body: string }[] = [];
      function recurse(dir: string) {
        for (const entry of readdirSync(dir, { withFileTypes: true })) {
          if (entry.isDirectory()) recurse(join(dir, entry.name));
          else if (entry.name.endsWith(".md")) {
            const full = join(dir, entry.name);
            const parsed = matter(readFileSync(full, "utf8"));
            out.push({ path: full, fm: parsed.data, body: parsed.content });
          }
        }
      }
      recurse(root);
      return out;
    }
